SMNS Basic ERD Diagram

This ERD shows the main entities and relationships in the Seminary Management and Results System.

erDiagram
    USERS {
        INT id PK
        VARCHAR username
        VARCHAR email
        VARCHAR password_hash
        ENUM role
        ENUM status
    }

    STUDENTS {
        INT id PK
        INT user_id FK
        VARCHAR student_id
        VARCHAR first_name
        VARCHAR last_name
        INT program_id FK
        INT level_year
        INT entry_year
    }

    LECTURERS {
        INT id PK
        INT user_id FK
        VARCHAR lecturer_id
        VARCHAR first_name
        VARCHAR last_name
    }

    ADMINS {
        INT id PK
        INT user_id FK
        VARCHAR first_name
        VARCHAR last_name
        VARCHAR email
    }

    FINANCE_STAFF {
        INT id PK
        INT user_id FK
        VARCHAR first_name
        VARCHAR last_name
        VARCHAR email
    }

    NOTIFICATIONS {
        INT id PK
        INT user_id FK
        VARCHAR title
        TEXT message
        ENUM type
    }

    PROGRAMS {
        INT id PK
        VARCHAR program_code
        VARCHAR program_name
        VARCHAR department
    }

    COURSES {
        INT id PK
        VARCHAR course_code
        VARCHAR course_name
        INT program_id FK
        INT level_year
    }

    ACADEMIC_YEARS {
        INT id PK
        VARCHAR year_name
        DATE start_date
        DATE end_date
    }

    SEMESTERS {
        INT id PK
        INT academic_year_id FK
        VARCHAR semester_name
        TINYINT semester_number
    }

    COURSE_REGISTRATIONS {
        INT id PK
        INT student_id FK
        INT course_id FK
        INT semester_id FK
        DATE registration_date
    }

    COURSE_ASSIGNMENTS {
        INT id PK
        INT lecturer_id FK
        INT course_id FK
        INT semester_id FK
        DATE assigned_date
    }

    RESULTS {
        INT id PK
        INT student_id FK
        INT course_id FK
        INT semester_id FK
        DECIMAL total_marks
        VARCHAR grade
        DECIMAL grade_points
    }

    STUDENT_GPAS {
        INT id PK
        INT student_id FK
        INT semester_id FK
        DECIMAL semester_gpa
        DECIMAL cumulative_gpa
    }

    GRADES {
        INT id PK
        VARCHAR grade_letter
        DECIMAL min_mark
        DECIMAL max_mark
        DECIMAL grade_point
    }

    INVOICES {
        INT id PK
        VARCHAR invoice_number
        INT student_id FK
        INT semester_id FK
        DECIMAL total_amount
        DECIMAL balance
    }

    PAYMENTS {
        INT id PK
        VARCHAR payment_id
        INT student_id FK
        INT invoice_id FK
        INT received_by FK
        DECIMAL amount
    }

    STUDENT_BALANCES {
        INT id PK
        INT student_id FK
        INT semester_id FK
        DECIMAL total_fees
        DECIMAL total_paid
        DECIMAL balance
    }

    FEES_STRUCTURE {
        INT id PK
        VARCHAR fee_name
        INT program_id FK
        INT semester_id FK
        DECIMAL amount
        INT level_year
    }

    USERS ||--|| STUDENTS : has
    USERS ||--|| LECTURERS : has
    USERS ||--|| ADMINS : has
    USERS ||--|| FINANCE_STAFF : has
    USERS ||--o{ NOTIFICATIONS : receives

    PROGRAMS ||--o{ STUDENTS : contains
    PROGRAMS ||--o{ COURSES : offers
    ACADEMIC_YEARS ||--o{ SEMESTERS : contains

    STUDENTS ||--o{ COURSE_REGISTRATIONS : makes
    COURSES ||--o{ COURSE_REGISTRATIONS : includes
    SEMESTERS ||--o{ COURSE_REGISTRATIONS : during

    LECTURERS ||--o{ COURSE_ASSIGNMENTS : assigned
    COURSES ||--o{ COURSE_ASSIGNMENTS : has
    SEMESTERS ||--o{ COURSE_ASSIGNMENTS : during

    STUDENTS ||--o{ RESULTS : gets
    COURSES ||--o{ RESULTS : produces
    SEMESTERS ||--o{ RESULTS : in
    GRADES ||..o{ RESULTS : classifies

    STUDENTS ||--o{ STUDENT_GPAS : has
    SEMESTERS ||--o{ STUDENT_GPAS : for

    STUDENTS ||--o{ INVOICES : billed
    SEMESTERS ||--o{ INVOICES : for
    INVOICES ||--o{ PAYMENTS : paid_by
    FINANCE_STAFF ||--o{ PAYMENTS : receives
    STUDENTS ||--o{ PAYMENTS : makes
    STUDENTS ||--o{ STUDENT_BALANCES : has
    SEMESTERS ||--o{ STUDENT_BALANCES : for

    PROGRAMS ||--o{ FEES_STRUCTURE : defines
    SEMESTERS ||--o{ FEES_STRUCTURE : applies_in
Arrow guide: `||--||` means one-to-one, `||--o{` means one-to-many, and dotted `||..o{` shows a logical relationship.
